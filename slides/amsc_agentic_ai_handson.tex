        \begin{itemize}\setlength\itemsep{1pt}
          \item[\faCheckCircle] Resolves account, queue, paths
          \item[\faCheckCircle] Stages the script and submits
          \item[\faCheckCircle] Monitors; diagnoses failures
          \item[\faCheckCircle] Reports back with evidence
        \end{itemize}
      \end{featbox}
    \end{column}
  \end{columns}

name: submit-job
description: Stage inputs, run an IRI batch
  job, monitor it, and report back.

2. globus_transfer, then poll until
   SUCCEEDED. A task ID is not a delivery.
3. get_system_status, then submit_job.

\begin{frame}[fragile]{Example 3: the Lab 02 MNIST trainer, unchanged, on a Polaris GPU}
  \begin{chatbubble}{labnavy}{\faUser\ You}
    \scriptsize ``Run my MNIST trainer from Lab 02 on one Polaris GPU node ---
    debug queue, 15 minutes, charged to my account. Stage it, submit it, watch
    it, and show me the final test accuracy it actually printed.''
  \end{chatbubble}
  \vspace{3pt}
  \begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.54\textwidth}
      \codetag{\faCode\ stage $\rightarrow$ submit $\rightarrow$ monitor $\rightarrow$ retrieve}
\begin{lstlisting}[style=pysnip]
globus_transfer(src="train.py",
  dst="/eagle/<project>/<you>/mnist/")
globus_transfer_status(task_id=...)  # SUCCEEDED
list_projects()              # resolve the account
submit_job(system="polaris", nodes=1,
           queue="debug", walltime="00:15:00",
           script_path=".../run.sh")
get_job_status(job_id=...)   # poll with backoff
read_file(path=".../job.out", tail=20)
\end{lstlisting}
    \end{column}
    \begin{column}{0.03\textwidth}\end{column}
    \begin{column}{0.43\textwidth}
      \sechead{accentteal}{\faRocket\ Same spine, bigger machine}
      \scriptsize
      \begin{itemize}\setlength\itemsep{1.5pt}
        \item[\faLaptop] The \emph{same} \texttt{train.py} Lab 02 wrote on your
          laptop. CPU to A100, no rewrite --- only \texttt{run.sh} is new.
        \item[\faFileCode] \texttt{run.sh} opens with \texttt{\#!/bin/bash -l}
          --- the login shell that puts \texttt{module} and conda on
          \texttt{PATH}. Without it the job dies at exit 127.
        \item[\faEye] A transfer task ID is not a delivery --- poll it to
          \texttt{SUCCEEDED} before the job reads the file.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
